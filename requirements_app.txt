# Vula! Print Label Printer - Desktop App Requirements
# For Debian 13 Trixie
#
# NOTE: PyQt6 and PyQt6-Qt6 are intentionally NOT listed here.
# They are installed as system packages (python3-pyqt6) via apt in
# install_printer_app.sh and inherited by the venv through
# --system-site-packages. Listing them here would cause pip to
# re-download 80+ MB of Qt6 binaries unnecessarily.
# On Windows, PyQt6 is not available as a system package — install it
# explicitly. The marker keeps pip from re-downloading it on Linux,
# where apt provides it via --system-site-packages.
PyQt6>=6.4; sys_platform == 'win32'

# Windows: pywin32 provides win32print (spooler submission — Phase B)
# and win32crypt (DPAPI for API-key-at-rest — Phase D). Linux gets
# neither, so the marker keeps pip from attempting the install there.
pywin32>=306; sys_platform == 'win32'

# HTTP Client
requests>=2.31.0

# Optional: better SSL certificate handling
certifi>=2023.0.0

# QR code generation for POS receipts.
# Required because old ESC/POS printers do not support the native GS ( k
# 2D-barcode command; we render QR codes as 1-bit raster images instead.
qrcode>=7.4.2

# Image manipulation for QR raster + logo raster generation.
pillow>=10.0.0

# Serial port communication for POS printers connected via USB-serial
# adapters (CH340, FTDI, PL2303, etc.). Needed because raw open() writes
# leave the port at an unconfigured baud rate, producing garbage output
# on the printer. pyserial negotiates termios correctly.
pyserial>=3.5
